cmake_minimum_required(VERSION 3.24)
project(BotWModsTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(mods_tests catalog_test.cc ../catalog.cpp ../../vfs/overlay.cpp)
target_compile_options(mods_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME mods_tests COMMAND mods_tests)
find_package(ZLIB REQUIRED)
add_executable(mods_online_tests online_test.cc ../json_min.cpp ../gamebanana.cpp ../zip_install.cpp ../catalog.cpp)
target_compile_options(mods_online_tests PRIVATE -Wall -Wextra -Werror)
target_link_libraries(mods_online_tests PRIVATE ZLIB::ZLIB)
add_test(NAME mods_online_tests COMMAND mods_online_tests)
add_executable(mods_store_tests store_test.cc ../store.cpp ../json_min.cpp ../gamebanana.cpp ../catalog.cpp)
target_compile_options(mods_store_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_definitions(mods_store_tests PRIVATE FIXTURE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME mods_store_tests COMMAND mods_store_tests)
add_executable(mods_local_tests local_test.cc ../store.cpp ../json_min.cpp ../gamebanana.cpp ../catalog.cpp ../../vfs/overlay.cpp)
target_compile_options(mods_local_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME mods_local_tests COMMAND mods_local_tests)
find_package(LibArchive QUIET)
if(LibArchive_FOUND)
add_executable(mods_archive_tests archive_test.cc ../archive_install.cpp ../zip_install.cpp)
target_compile_options(mods_archive_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_compile_definitions(mods_archive_tests PRIVATE BOTW_MOD_LIBARCHIVE=1)
target_link_libraries(mods_archive_tests PRIVATE LibArchive::LibArchive ZLIB::ZLIB)
add_test(NAME mods_archive_tests COMMAND mods_archive_tests)
else()
  message(STATUS "libarchive headers unavailable; optional 7z/RAR backend tests skipped")
endif()
find_package(Threads REQUIRED)
add_executable(mods_store_view_tests store_view_test.cc ../store.cpp ../json_min.cpp ../gamebanana.cpp)
target_compile_options(mods_store_view_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME mods_store_view_tests COMMAND mods_store_view_tests)
target_link_libraries(mods_store_view_tests PRIVATE Threads::Threads)
