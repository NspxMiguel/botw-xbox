cmake_minimum_required(VERSION 3.20)
project(BotWOverlayTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(overlay_test ../overlay.cpp overlay_test.cc)
target_compile_features(overlay_test PRIVATE cxx_std_20)
target_link_libraries(overlay_test PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(overlay_test PRIVATE /W4 /UNDEBUG /utf-8)
else()
  target_compile_options(overlay_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME overlay COMMAND overlay_test)
add_executable(hle_overlay_test ../overlay.cpp ../../coreinit/fs.cpp hle_test.cc)
target_include_directories(hle_overlay_test PRIVATE ../.. ../../../include)
target_compile_features(hle_overlay_test PRIVATE cxx_std_20)
if(MSVC)
  target_compile_options(hle_overlay_test PRIVATE /UNDEBUG /utf-8)
else()
  target_compile_options(hle_overlay_test PRIVATE -UNDEBUG)
endif()
target_link_libraries(hle_overlay_test PRIVATE Threads::Threads)
add_test(NAME hle_overlay COMMAND hle_overlay_test)
