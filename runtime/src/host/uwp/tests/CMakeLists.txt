cmake_minimum_required(VERSION 3.24)
project(BotWUwpHostTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(uwp_host_tests tests.cc ../controller.cpp ../launcher.cpp)
target_compile_options(uwp_host_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME uwp_host_tests COMMAND uwp_host_tests)
add_executable(uwp_art_tests art_tests.cc)
target_compile_options(uwp_art_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME uwp_art_tests COMMAND uwp_art_tests)
add_executable(uwp_drive_sound_tests drive_sound_test.cc)
target_compile_options(uwp_drive_sound_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME uwp_drive_sound_tests COMMAND uwp_drive_sound_tests)
find_package(ZLIB REQUIRED)
find_path(ZSTD_INCLUDE_DIR zstd.h HINTS /opt/homebrew/include REQUIRED)
find_library(ZSTD_LIBRARY zstd HINTS /opt/homebrew/lib REQUIRED)
add_executable(uwp_setup_tests setup_tests.cc ../launcher.cpp)
target_include_directories(uwp_setup_tests PRIVATE ${ZSTD_INCLUDE_DIR})
target_link_libraries(uwp_setup_tests PRIVATE ZLIB::ZLIB ${ZSTD_LIBRARY})
target_compile_options(uwp_setup_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME uwp_setup_tests COMMAND uwp_setup_tests)
add_executable(uwp_polish_tests polish_tests.cc)
target_compile_options(uwp_polish_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME uwp_polish_tests COMMAND uwp_polish_tests)
add_executable(uwp_diagnostic_tests diagnostic_tests.cc)
target_include_directories(uwp_diagnostic_tests PRIVATE ${ZSTD_INCLUDE_DIR})
target_link_libraries(uwp_diagnostic_tests PRIVATE ZLIB::ZLIB ${ZSTD_LIBRARY})
target_compile_options(uwp_diagnostic_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME uwp_diagnostic_tests COMMAND uwp_diagnostic_tests)
