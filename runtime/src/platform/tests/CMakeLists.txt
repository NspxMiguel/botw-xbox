cmake_minimum_required(VERSION 3.24)
project(BotWPlatformTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
enable_testing()
if(NOT APPLE)
  message(FATAL_ERROR "This harness exercises the Apple backend on macOS")
endif()
add_executable(platform_tests platform_test.cc ../clock.cpp ../apple/platform.cpp)
target_compile_options(platform_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME platform_tests COMMAND platform_tests)
add_executable(sim_series_s_tests sim_series_s_test.cc ../../sim_series_s.cpp)
target_compile_options(sim_series_s_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME sim_series_s_off COMMAND sim_series_s_tests off)
set_tests_properties(sim_series_s_off PROPERTIES ENVIRONMENT "BOTW_SIM_SERIES_S=0;BOTW_SIM_GPU_FACTOR=3;BOTW_SIM_CPU_FACTOR=1.05")
add_test(NAME sim_series_s_on COMMAND sim_series_s_tests on)
set_tests_properties(sim_series_s_on PROPERTIES ENVIRONMENT "BOTW_SIM_SERIES_S=1;BOTW_SIM_GPU_FACTOR=3;BOTW_SIM_CPU_FACTOR=1.05")
add_test(NAME sim_series_s_invalid COMMAND sim_series_s_tests off)
set_tests_properties(sim_series_s_invalid PROPERTIES ENVIRONMENT "BOTW_SIM_SERIES_S=true;BOTW_SIM_GPU_FACTOR=nan;BOTW_SIM_CPU_FACTOR=2.01")
add_executable(fsr_test fsr_test.cc)
target_compile_options(fsr_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
add_test(NAME fsr_test COMMAND fsr_test)
